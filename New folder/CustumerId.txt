import java.util.*;

class CustomerID
{
	public static int customerId(int num[],int key)
	{
		for(int i=0;i<num.length;i++)
		{
			if(num[i]==key)
			{
				return i;
			}
		
		}
		return-1;
	
		
	}
	public static void main(String s[])
	{
		Scanner sc=new Scanner(System.in);
		System.out.print("Enter the number of customers:");
		int n=sc.nextInt();
		int A[]=new int[n];

		System.out.print("Enter the Customer id:");

		for(int i=0;i<A.length;i++)
		{
			A[i]=sc.nextInt();
		}
		System.out.print("Customer id are displayed as:");
		

		for(int i=0;i<A.length;i++)
		{
			System.out.print(A[i]+"\t");
		}

		System.out.print("Customer id to find:");
		int x=sc.nextInt();

		int result=customerId(A,x);
		if(result==-1)
		{
			System.out.print("Customer id\t"+x+"\t not found");
		}

		else
		{
			System.out.print("Customer id\t"+x+"\t is found at:"+result);

		}

		sc.close();

	}
}
